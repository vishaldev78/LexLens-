    98→function ReportInner() {
    99→  const { id } = useParams<{ id: string }>();
   100→  const router = useRouter();
   101→  const { t, locale } = useLang();
   102→  const tt = t as unknown as TT;
   103→
   104→  const [detail, setDetail] = useState<Detail | null>(null);
   105→  const [failed, setFailed] = useState(false);
   106→  const [notFound, setNotFound] = useState(false);
   107→
   108→  const [savedFlash, setSavedFlash] = useState("");
   109→  const [receiptInput, setReceiptInput] = useState("");
   110→  const [receiptBusy, setReceiptBusy] = useState(false);
   111→  const [retryBusy, setRetryBusy] = useState(false);
   112→  const [draftBusy, setDraftBusy] = useState(false);
   113→  const [draftErr, setDraftErr] = useState("");
   114→  const [evidenceBusy, setEvidenceBusy] = useState(false);
   115→  const [briefOpen, setBriefOpen] = useState(false);
   116→  const [showOriginal, setShowOriginal] = useState(false);
   117→  const [copied, setCopied] = useState(false);
   118→
   119→  const inputTimers = useRef<Record<string, ReturnType<typeof setTimeout>>>({});
   120→
   121→  const load = useCallback(async () => {
   122→    try {
   123→      const res = await fetch(`/api/notices/${id}`, { cache: "no-store" });
   124→      if (res.status === 404) {
   125→        setNotFound(true);
   126→        return;
   127→      }
   128→      if (!res.ok) throw new Error(String(res.status));
   129→      setDetail((await res.json()) as Detail);
   130→    } catch {
   131→      setFailed(true);
   132→    }
   133→  }, [id]);
   134→
   135→  useEffect(() => {
   136→    void load();
   137→  }, [load]);
   138→
   139→  const today = todayISO();
   140→  const view = useMemo(
   141→    () => (detail?.base ? buildCaseView(detail.base, detail.userState, locale, today) : null),
   142→    [detail, locale, today],
   143→  );
   144→
   145→  function flash(key: string) {
   146→    setSavedFlash(key);
   147→    setTimeout(() => setSavedFlash(""), 2500);
   148→  }
   149→
   150→  /** Patch the workspace record, then refresh the server-computed summary. */
   151→  async function patchNotice(body: Record<string, unknown>): Promise<boolean> {
   152→    try {
   153→      const res = await fetch(`/api/notices/${id}`, {
   154→        method: "PATCH",
   155→        headers: { "Content-Type": "application/json" },
   156→        body: JSON.stringify(body),
   157→      });
   158→      if (!res.ok) return false;
   159→      await load();
   160→      return true;
   161→    } catch {
   162→      return false;
   163→    }
   164→  }
   165→
   166→  function saveReceipt(raw: string) {
   167→    const iso = /^\d{4}-\d{2}-\d{2}$/.test(raw) ? raw : parseHumanDate(raw);
   168→    if (!iso) return;
   169→    setReceiptBusy(true);
   170→    void patchNotice({ receiptDate: iso }).then((ok) => {
   171→      setReceiptBusy(false);
   172→      if (ok) {
   173→        setReceiptInput("");
   174→        flash("receipt_saved");
   175→      }
   176→    });
   177→  }
   178→
   179→  function setInput(field: string, kind: "date" | "text" | "number", raw: string) {
   180→    if (!detail) return;
   181→    const inputs = { ...detail.userState.inputs };
   182→    if (!raw.trim()) {
   183→      delete inputs[field];
   184→      setDetail({ ...detail, userState: { ...detail.userState, inputs } });
   185→    } else if (kind === "date") {
   186→      const iso = /^\d{4}-\d{2}-\d{2}$/.test(raw) ? raw : parseHumanDate(raw);
   187→      if (!iso) return; // invalid dates are never stored
   188→      inputs[field] = { value: raw, iso, num: null, at: Date.now() };
   189→      setDetail({ ...detail, userState: { ...detail.userState, inputs } });
   190→    } else if (kind === "number") {
   191→      const num = Number(raw.replace(/[^\d.]/g, ""));
   192→      inputs[field] = { value: raw, iso: null, num: Number.isFinite(num) ? num : null, at: Date.now() };
   193→      setDetail({ ...detail, userState: { ...detail.userState, inputs } });
   194→    } else {
   195→      inputs[field] = { value: raw, iso: null, num: null, at: Date.now() };
   196→      setDetail({ ...detail, userState: { ...detail.userState, inputs } });
   197→    }
   198→    flash(field);
   199→    clearTimeout(inputTimers.current[field]);
   200→    inputTimers.current[field] = setTimeout(() => {
   201→      void patchNotice({ inputs: detail.userState.inputs });
   202→    }, 600);
   203→  }
   204→
   205→  function setPosition(p: UserPosition) {
   206→    if (!detail) return;
   207→    setDetail({ ...detail, userState: { ...detail.userState, position: p } });
   208→    void patchNotice({ position: p });
   209→  }
   210→
   211→  async function addEvidence(files: FileList | null) {
   212→    if (!files?.length || !detail) return;
   213→    setEvidenceBusy(true);
   214→    try {
   215→      const form = new FormData();
   216→      Array.from(files)
   217→        .slice(0, 6)
   218→        .forEach((f) => form.append("files", f));
   219→      const res = await fetch(`/api/notices/${id}/evidence`, { method: "POST", body: form });
   220→      if (res.ok) {
   221→        await load();
   222→        flash("evidence");
   223→      }
   224→    } finally {
   225→      setEvidenceBusy(false);
   226→    }
   227→  }
   228→
   229→  async function generateDraft() {
   230→    if (!detail || !view) return;
   231→    setDraftBusy(true);
   232→    setDraftErr("");
   233→    try {
   234→      const res = await fetch(`/api/notices/${id}/draft`, {
   235→        method: "POST",
   236→        headers: { "Content-Type": "application/json" },
   237→        body: JSON.stringify({ locale }),
   238→      });
   239→      const data = (await res.json()) as { draft?: string; source?: "ai" | "template"; error?: string };
   240→      if (!data.draft) throw new Error(data.error ?? "draft failed");
   241→      setDetail({ ...detail, userState: { ...detail.userState, draft: { text: data.draft, source: data.source ?? "template", at: Date.now() } } });
   242→    } catch {
   243→      const fallback = templateDraft(view, locale);
   244→      if (fallback) {
   245→        setDetail({ ...detail, userState: { ...detail.userState, draft: { text: fallback, source: "template", at: Date.now() } } });
   246→      } else {
   247→        setDraftErr(tt.rd_error ?? "Could not prepare the draft. Please try again.");
   248→      }
   249→    } finally {
   250→      setDraftBusy(false);
   251→    }
   252→  }
   253→
   254→  const brief = useMemo(() => (view ? buildBrief(view, locale) : null), [view, locale]);
   255→
   256→  function openBrief() {
   257→    setBriefOpen(true);
   258→    // Persist the brief for future sessions (fire-and-forget).
   259→    void fetch(`/api/notices/${id}/brief`, {
   260→      method: "POST",
   261→      headers: { "Content-Type": "application/json" },
   262→      body: JSON.stringify({ locale }),
   263→    }).catch(() => {});
   264→  }
   265→
   266→  async function retryAnalysis() {
   267→    setRetryBusy(true);
   268→    try {
   269→      const res = await fetch(`/api/notices/${id}/analyze`, { method: "POST" });
   270→      if (res.ok) await load();
   271→    } finally {
   272→      setRetryBusy(false);
   273→    }
   274→  }
   275→
   276→  async function remove() {
   277→    if (!window.confirm(tt.nl_confirm)) return;
   278→    const res = await fetch(`/api/notices/${id}`, { method: "DELETE" });
   279→    if (res.ok) router.push("/analyze");
   280→  }
   281→
   282→  function downloadFile(name: string, content: string, mime = "text/plain") {
   283→    const blob = new Blob([content], { type: mime });
   284→    const url = URL.createObjectURL(blob);
   285→    const a = document.createElement("a");
   286→    a.href = url;
   287→    a.download = name;
   288→    a.click();
   289→    URL.revokeObjectURL(url);
   290→  }
   291→
   292→  async function copySummary() {
   293→    if (!view || !detail?.base) return;
   294→    const b = detail.base.localized[locale] ?? detail.base.localized.en;
   295→    const dl = view.deadlines.find((d) => d.status === "calculated");
   296→    const lines = [
   297→      "LexLens — Case summary",
   298→      "──────────────────────────",
   299→      b.summary,
   300→      "",
   301→      `${tt.ac_deadline_exact}: ${dl ? `${dl.deadline} (${dl.label})` : tt.ac_cannot_calc}`,
   302→      "",
   303→      `${tt.ac_next}:`,
   304→      ...view.actions.map((a, i) => `${i + 1}. ${a.title}`),
   305→      "",
   306→      tt.rs_disclaimer,
   307→    ];
   308→    try {
   309→      await navigator.clipboard.writeText(lines.join("\n"));
   310→      setCopied(true);
   311→      setTimeout(() => setCopied(false), 2000);
   312→    } catch {
   313→      /* clipboard blocked */
   314→    }
   315→  }
   316→
   317→  /* ── states ── */
   318→  if (notFound) {
   319→    return (
   320→      <div className="mx-auto max-w-xl px-4 py-20 text-center">
   321→        <FileText className="mx-auto h-10 w-10 text-slate-300" />
   322→        <h1 className="mt-4 text-xl font-bold text-slate-900">{tt.rs_notfound_t}</h1>
   323→        <p className="mt-2 text-sm leading-relaxed text-slate-600">{tt.rs_notfound_d}</p>
   324→        <Link href="/analyze" className="mt-6 inline-flex items-center gap-2 rounded-xl bg-indigo-600 px-5 py-3 text-sm font-bold text-white shadow-sm hover:bg-indigo-700">
   325→          {tt.nav_analyze}
   326→        </Link>
   327→      </div>
   328→    );
   329→  }
   330→  if (failed) {
   331→    return (
   332→      <div className="mx-auto max-w-md px-4 py-24 text-center">
   333→        <AlertTriangle className="mx-auto h-10 w-10 text-amber-500" />
   334→        <h1 className="mt-4 text-lg font-bold text-slate-900">{tt.cm_error}</h1>
   335→      </div>
   336→    );
   337→  }
   338→  if (!detail) {
   339→    return (
   340→      <div className="flex min-h-[50vh] items-center justify-center">
   341→        <div className="h-8 w-8 animate-spin rounded-full border-4 border-indigo-200 border-t-indigo-600" />
   342→      </div>
   343→    );
   344→  }
   345→
   346→  const n = detail.notice;
   347→  const base = detail.base;
   348→  const hasReport = !!base && (n.analysisStatus === "READY" || !!base);
   349→
   350→  return (
   351→    <div className="relative">
   352→      <div className="pointer-events-none absolute inset-x-0 top-0 -z-10 h-72 bg-gradient-to-b from-indigo-50/80 to-transparent" />
   353→      <div className="mx-auto max-w-4xl px-4 py-10 sm:px-6 sm:py-14">
   354→        {!hasReport ? (
   355→          <PendingCard n={n} tt={tt} retryBusy={retryBusy} onRetry={() => void retryAnalysis()} onRemove={() => void remove()} />
   356→        ) : (
   357→          <>
   358→            <ReportBody
   359→              detail={detail}
   360→              view={view!}
   361→              tt={tt}
   362→              locale={locale}
   363→              brief={brief!}
   364→              flags={{
   365→                savedFlash,
   366→                receiptInput,
   367→                setReceiptInput,
   368→                receiptBusy,
   369→                retryBusy,
   370→                draftBusy,
   371→                draftErr,
   372→                evidenceBusy,
   373→                briefOpen,
   374→                setBriefOpen,
   375→                showOriginal,
   376→                setShowOriginal,
   377→                copied,
   378→              }}
   379→              actions={{
   380→                saveReceipt,
   381→                setInput,
   382→                setPosition,
   383→                addEvidence,
   384→                generateDraft,
   385→                openBrief,
   386→                retryAnalysis: () => void retryAnalysis(),
   387→                remove: () => void remove(),
   388→                patchNotice,
   389→                downloadFile,
   390→                copySummary: () => void copySummary(),
   391→                printBrief: () => {
   392→                  document.body.classList.add("brief-print");
   393→                  setTimeout(() => {
   394→                    window.print();
   395→                    setTimeout(() => document.body.classList.remove("brief-print"), 400);
   396→                  }, 60);
   397→                },
   398→              }}
   399→            />
   400→          </>
   401→        )}
   402→      </div>
   403→    </div>
   404→  );
   405→}

   409→function PendingCard({
   410→  n,
   411→  tt,
   412→  retryBusy,
   413→  onRetry,
   414→  onRemove,
   415→}: {
   416→  n: NoticeSummary;
   417→  tt: TT;
   418→  retryBusy: boolean;
   419→  onRetry: () => void;
   420→  onRemove: () => void;
   421→}) {
   422→  const failed = n.analysisStatus === "FAILED";
   423→  return (
   424→    <div className="rounded-2xl border border-slate-200 bg-white p-8 text-center shadow-sm">
   425→      <span
   426→        className={`mx-auto flex h-14 w-14 items-center justify-center rounded-2xl ${
   427→          failed ? "bg-red-50 text-red-500" : "bg-indigo-50 text-indigo-500"
   428→        }`}
   429→      >
   430→        {failed ? <AlertTriangle className="h-7 w-7" /> : <Loader2 className="h-7 w-7 animate-spin" />}
   431→      </span>
   432→      <h1 className="mt-4 text-xl font-extrabold text-slate-900">{failed ? tt.rp_status_failed : tt.rp_status}</h1>
   433→      <p className="mx-auto mt-2 max-w-md text-sm leading-relaxed text-slate-600">{failed ? tt.nl_retry : tt.nl_empty_d}</p>
   434→      <div className="mt-6 flex flex-wrap items-center justify-center gap-2.5">
   435→        <button
   436→          onClick={onRetry}
   437→          disabled={retryBusy}
   438→          className="inline-flex items-center gap-2 rounded-xl bg-indigo-600 px-5 py-3 text-sm font-bold text-white shadow-sm hover:bg-indigo-700 disabled:opacity-60"
   439→        >
   440→          {retryBusy ? <Loader2 className="h-4 w-4 animate-spin" /> : <RotateCcw className="h-4 w-4" />}
   441→          {failed ? tt.nl_retry : tt.rp_retry}
   442→        </button>
   443→        <button
   444→          onClick={onRemove}
   445→          className="inline-flex items-center gap-2 rounded-xl border border-slate-300 bg-white px-5 py-3 text-sm font-bold text-red-600 hover:border-red-300 hover:bg-red-50"
   446→        >
   447→          <Trash2 className="h-4 w-4" /> {tt.rp_del}
   448→        </button>
   449→      </div>
   450→    </div>
   451→  );
   452→}

   472→interface Actions {
   473→  saveReceipt: (raw: string) => void;

   475→  setPosition: (p: UserPosition) => void;
   476→  addEvidence: (files: FileList | null) => Promise<void>;

   487→function ReportBody({
   488→  detail,
   489→  view,
   490→  tt,
   491→  locale,
   492→  brief,
   493→  flags,
   494→  actions,
   495→}: {
   496→  detail: Detail;
   497→  view: NonNullable<ReturnType<typeof buildCaseView>>;
   498→  tt: TT;
   499→  locale: Locale;
   500→  brief: ReturnType<typeof buildBrief>;
   501→  flags: Flags;
   502→  actions: Actions;
   503→}) {
   504→  const { notice: n } = detail;
   505→  const base = detail.base!;
   506→  const sev = base.severity.level;
   507→  const meta = view.deadlines.find((d) => d.status === "calculated");
   508→  const dlLabels = { noDate: tt.rs_no_date, overdue: tt.rs_overdue, today: tt.rs_today, oneLeft: tt.rs_1_day_left, daysLeft: tt.rs_days_left, cs_overdue: tt.cs_overdue, rdl_urgency_critical: tt.rdl_urgency_critical, rdl_none: tt.rdl_none, rs_days_left: tt.rs_days_left };
   509→
   510→  const [completedBusy, setCompletedBusy] = useState(false);
   511→
   512→  async function toggleCompleted() {
   513→    setCompletedBusy(true);
   514→    await actions.patchNotice({ completed: !n.completed });
   515→    setCompletedBusy(false);
   516→  }
   517→
   518→  return (
   519→    <>
   520→      {/* header */}
   521→      <div className="flex flex-col gap-4 sm:flex-row sm:items-end sm:justify-between">
   522→        <div className="min-w-0">
   523→          <span className="inline-flex items-center gap-1.5 rounded-full border border-indigo-200 bg-indigo-50 px-3 py-1 text-[11px] font-bold uppercase tracking-wider text-indigo-700">
   524→            <ClipboardList className="h-3 w-3" /> {tt.rs_kicker}
   525→          </span>
   526→          <h1 className="mt-3 text-2xl font-extrabold tracking-tight text-slate-900 sm:text-3xl">{view.type_label}</h1>
   527→          <p className="mt-1 truncate text-sm text-slate-500">
   528→            {n.title} · {view.jurisdiction_label}
   529→          </p>
   530→          {/* PRD §14 — deterministic classification (never "debt collection" for a §138 notice) */}
   531→          <p className="mt-1.5 text-[13px] font-medium text-slate-600">
   532→            {tt.cls_primary}: <span className="font-bold text-slate-800">{view.classification.primary}</span>
   533→            {view.classification.subcategory && (
   534→              <>
   535→                {" · "}{tt.cls_sub}: <span className="font-bold text-slate-800">{view.classification.subcategory}</span>
   536→              </>
   537→            )}
   538→            {view.classification.secondary && (
   539→              <>
   540→                {" · "}{tt.cls_secondary}: <span className="text-slate-600">{view.classification.secondary}</span>
   541→              </>
   542→            )}
   543→          </p>
