    94→  function mutate(patch: Partial<UserCaseState>) {
    95→    setRecord((r) => {
    96→      if (!r) return r;
    97→      const user = { ...r.user, ...patch };
    98→      if (r.id) patchCase(r.id, () => user);
    99→      return { ...r, user };
   100→    });

   103→  function setInput(field: string, kind: "date" | "text" | "number", raw: string) {
   104→    if (!raw.trim()) {
   105→      const inputs = { ...record?.user.inputs };
   106→      delete inputs[field];
   107→      mutate({ inputs });
   108→      return;
   109→    }
   110→    if (kind === "date") {
   111→      const iso = /^\d{4}-\d{2}-\d{2}$/.test(raw) ? raw : parseHumanDate(raw);
   112→      if (!iso) return; // invalid dates are never stored
   113→      mutate({ inputs: { ...record?.user.inputs, [field]: { value: raw, iso, num: null, at: Date.now() } } });
   114→    } else if (kind === "number") {
   115→      const num = Number(raw.replace(/[^\d.]/g, ""));
   116→      mutate({ inputs: { ...record?.user.inputs, [field]: { value: raw, iso: null, num: Number.isFinite(num) ? num : null, at: Date.now() } } });
   117→    } else {
   118→      mutate({ inputs: { ...record?.user.inputs, [field]: { value: raw, iso: null, num: null, at: Date.now() } } });
   119→    }
   120→    setSavedFlash(field);
   121→    setTimeout(() => setSavedFlash(""), 2500);
   122→  }

   124→  function addEvidence(files: FileList | null) {
   125→    if (!files?.length) return;

   127→      id: `${Date.now()}_${i}`,

   131→      addedAt: Date.now(),
   132→    }));
   133→    mutate({ evidence: [...(record?.user.evidence ?? []), ...ups] });
   134→  }

   136→  async function generateDraft() {
   137→    if (!record) return;
   138→    setDraftBusy(true);
   139→    setDraftErr("");
   140→    try {
   141→      const res = await fetch("/api/draft", {
   142→        method: "POST",
   143→        headers: { "Content-Type": "application/json" },
   144→        body: JSON.stringify({
   145→          notice_text: record.noticeText ?? "",
   146→          notice_type: record.base.notice_type,
   147→          locale,
   148→          position: record.user.position,
   149→          missing: view?.missing.map((m) => m.field) ?? [],
   150→          deadline_line: view?.deadlines.find((d) => d.status === "calculated")
   151→            ? `${view.deadlines[0].label}: ${view.deadlines[0].deadline}`
   152→            : null,
   153→          facts: view?.facts.filter((f) => f.value).map((f) => ({ key: f.key, value: f.value })),
   154→        }),
   155→      });
   156→      const data = (await res.json()) as { draft?: string; source?: "ai" | "template"; error?: string };
   157→      if (!data.draft) throw new Error(data.error ?? "draft failed");
   158→      mutate({ draft: { text: data.draft, source: data.source ?? "template", at: Date.now() } });
   159→    } catch (err) {
   160→      const fallback = view ? templateDraft(view, locale) : "";
   161→      if (fallback) mutate({ draft: { text: fallback, source: "template", at: Date.now() } });
   162→      else setDraftErr(err instanceof Error ? err.message : String(err));
   163→    } finally {
   164→      setDraftBusy(false);
   165→    }
   166→  }

   174→    a.click();
   175→    URL.revokeObjectURL(url);
